Create function Zvonki(lastname varchar(255))
Returns varshar
as $$
`
	select P1.Name
	from Phone p1
	join call c
	on p1.phone = c.c_in
	join phone p2
	on p2.phone = c.c_out
	where p2.name = $1
	group by p1.name
	order by count(*) desc
	limit 1;
end
` language sql;

create function bb(varchar, varchar)
returns setof date
as $$
	declare a varchar, b varchar
	select phone into (as) a from phones where name = $1;
	select phone into (as) b from phones where name = $2;
	select c_date from calls
	where c_out = a and c_in = b;
end
$$ language sql;

Create function Zvonki(lastname varchar(255))
Returns Table(AName varchar, Aphone varchar)
as $$
`
	select P1.Name, count(*)
	from Phone p1
	join call c
	on p1.phone = c.c_in
	join phone p2
	on p2.phone = c.c_out
	where p2.name = $1
	group by p1.name;
end
` language sql;
