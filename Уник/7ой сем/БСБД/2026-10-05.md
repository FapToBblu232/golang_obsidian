CREATE OR REPLACE PROCEDURE schema.proc_name(
    p_id      bigint,               -- IN по умолчанию
    p_name    text DEFAULT 'x',     -- со значением по умолчанию
    INOUT p_result text DEFAULT NULL -- выходной параметр
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_count integer := 0;           -- локальные переменные
    v_row   users%ROWTYPE;          -- переменная под строку таблицы
BEGIN
    -- основная логика
    SELECT * INTO v_row FROM users WHERE id = p_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'User % not found', p_id;
    END IF;

    UPDATE users SET name = p_name WHERE id = p_id;
    GET DIAGNOSTICS v_count = ROW_COUNT;

    p_result := format('updated %s rows', v_count);
END;
$$;

Create or replace procedure swap(int, date)
language sql
as $$
Begin
	insert into Проверка(id_счета, Сумма, Дата-операции, Статус)
	select id_счета, Сумма, $2, 'Расход'
	from Расход
	where id-менеджера = $1
	and Дата-операции = $2
	Union
	select id_счета, Сумма, $2, 'Приход'
	from Приход
	where id-менеджера = $1
	and Дата-операции = $2;
	
	delete from Расход
	where id-менеджера = $1
	and Дата-операции = $2;
	delete from Приход
	where id-менеджера = $1
	and Дата-операции = $2;
end;
$$;
