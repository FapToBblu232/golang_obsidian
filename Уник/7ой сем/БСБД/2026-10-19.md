### Семинар
```
create trigger Aa
Before Update of id_читателя
on Читатели
for each row
execute function Af()

create or replace function Af()
returns trigger
begin
	update Дети
	set id_Читателя = New.id_читателя
	where id_Читателя = OLD.id_Читателя
	
	update Взрослые
	set id_Читателя = New.id_читателя
	where id_Читателя = OLD.id_Читателя
	
	update Дети
	set id_род = New.id_читателя
	where id_Читателя = OLD.id_Читателя
	
	return NEW;
end;

Language SQL;

`raise exception = return null`

```
create trigger Aa
befote Insert
on Заказы
for each row
execute function Af();

create or replace function Af()
returns trigger
begin
declare a varchar
	if (select Общее_кол-во from Продукты
		where id_продукта = NEW.id_продукта) < New.количество
	then
		set a = (select название from Продукты
			where id_продукта = NEW.id_продукта);
		raise Exception 'Не хватает %', a;
	end if;
	update Продукты
	set общее кол-во = общее кол-во - New.количество
	where id_продукта = NEW.id_продукта;
	return NEW;
end;
language sql;
